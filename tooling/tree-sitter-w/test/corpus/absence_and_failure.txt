fn firstOpen(orders: ref Array<Order>): ref Order? {

  return guest?.profile?.name ?? storedName ?? "Anonymous"
}

fn load(source: ref String): Order throws AppError {
  let result = Result.capture(() => try parse(source))
  let cached = try? parse(cacheSource)
  let order = try result

  do {
    return try validate(order)
  } catch .invalid(let token) if token.isRecoverable {
    return try repair(order)
  } catch error {
    throw .validation(error)
  }
}

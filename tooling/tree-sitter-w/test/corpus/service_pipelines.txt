async fn warm(
  ovens: ref ServiceRef<OvenApi>,
  target: Temperature,
  duration: Duration,
): (ServiceRef<OvenLeaseApi>, OvenReady) throws KitchenError {
  let (lease, ready) = try await pipeline {
    let lease = ovens.acquire(target, duration: duration)
    let ready = lease.preheat()
    commit (lease, ready)
  }

  return (lease, ready)
}

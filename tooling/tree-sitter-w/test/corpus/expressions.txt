fn calculate(
  a: Int,
  b: Int,
  c: Int,
  shift: UInt,
  minimum: Int,
  maximum: Int,
  primary: Int?,
  secondary: Int?,
  payload: ref any Reflectable,
  stage: ServiceStage,
) {
  let power = -2.0 ** 2 ** -3
  let arithmetic = a + b * c << shift
  let ordered = minimum <= a && a < maximum
  let typed = payload is ReservationKey
  let recovered = payload as? ReservationKey
  let staticId = type of ReservationKey
  let dynamicId = type of payload
  let staticInfo = info of ReservationKey
  let dynamicInfo = info of payload
  let queryPrecedence = type of value == other
  let groupedQuery = type of (value == other)
  let preparing = stage is .preparing
  let fallback = primary ?? secondary ?? computeFallback()
  registers[index()] <<= 1
}

async fn execute(
  store: ref Store,
  ovens: ref ServiceRef<OvenApi>,
) throws ExecuteError {
  let handle = unsafe {
    openHandle()
  }

  let ready = try await pipeline {
    let lease = ovens.acquire()
    let ready = lease.preheat()
    commit ready
  }

  let receipt = try await pipeline<transaction: {
    isolation: .serializable,
    access: .readWrite,
  }> tx = store {
    let value = try await tx.read()
    commit value
  }

  close(handle, ready: ready, receipt: receipt)
}

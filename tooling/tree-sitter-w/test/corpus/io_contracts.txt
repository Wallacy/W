==================
File rights and positional read
==================

async fn readBlock(
  files: ref FileSystem,
  path: ref Path,
  offset: FileOffset,
): Bytes throws IoError {
  let file = try await files.open<[.read]>(path)
  var bytes = Bytes()
  let _ = try await file.read(
    at: offset,
    appendTo: inout bytes,
    maximum: 4096,
  )
  return bytes
}

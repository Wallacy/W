==================
Generic protocols and enum subsets
==================

protocol Source<Item> {
  fn first(): ref Item?
}

protocol Counted {
  fn count(): usize
}

protocol Catalog<Item>: Source<Item> & Counted {}

struct Shelf<T> {
  values: Array<T>
}

extension<T: Display & Equatable> Shelf<T>: Catalog {
  alias Item = T

  fn first(): ref T? {
    return values.first
  }

  fn count(): usize {
    return values.count
  }
}

enum Stage {
  accepted
  preparing
  serving
  cancelled
}

alias WorkStage = Stage<[.preparing, .serving]>

enum Outcome<T> {
  ready(T)
  delayed(u32)
  cancelled
}

alias Continuing<T> = Outcome<T><[.ready, .delayed]>
alias Ready<T> = Outcome<T><[.ready]>

enum Fault: Error {
  delayed(u32)
  universeEnded
}

alias RecoverableFault = Fault<[.delayed]>

struct StagePath<const _ stages: StaticList<Stage>> {
  id: usize
}

struct Tile<const rows: usize, const columns: usize> {
  pixels: [u8; rows * columns]
}

fn render<T: Display & Equatable, S: Source<T>>(source: ref S): String? {
  guard let ref item = source.first() else return .none
  return item.display()
}

fn reserve(): WorkStage throws RecoverableFault {
  return .preparing
}

==================
Const enum typestate and specialized extensions
==================

enum Mode {
  idle
  ready
}

struct Session<const _ state: Mode> {
  id: u16
}

enum Activation {
  ready(Session<.ready>)
}

extension Session<.idle> {
  take fn activate(): Activation {
    return .ready(Session<.ready>(id: id))
  }
}

protocol Source<Item> {
  fn first(): ref Item?
}

protocol Counted {
  fn count(): usize
  fn isEmpty(): Bool
}

extension Counted {
  fn isEmpty(): Bool {
    return count() == 0
  }
}

protocol Catalog<Item>: Source<Item> & Counted {}

struct Shelf<T> {
  values: Array<T>
}

extension<T: Display & Equatable> Shelf<T>: Catalog {
  alias Item = T

  fn first(): ref T? {
    return values.first
  }

  fn count(): usize {
    return values.count
  }
}

enum Stage {
  accepted
  preparing
  serving
  cancelled
}

alias WorkStage = Stage<[.preparing, .serving]>

enum Outcome<T> {
  ready(T)
  delayed(u32)
  cancelled
}

alias Continuing<T> = Outcome<T><[.ready, .delayed]>
alias Ready<T> = Outcome<T><[.ready]>

enum Fault: Error {
  delayed(u32)
  universeEnded
}

alias RecoverableFault = Fault<[.delayed]>

struct StagePath<_ stages: StaticList<Stage>> {
  id: usize
}

struct Tile<rows: usize, columns: usize> {
  pixels: [u8; rows * columns]
}

fn render<T: Display & Equatable, S: Source<T>>(source: ref S): String? {
  guard let ref item = source.first() else return .none
  return item.display()
}

fn reserve(): WorkStage throws RecoverableFault {
  return .preparing
}

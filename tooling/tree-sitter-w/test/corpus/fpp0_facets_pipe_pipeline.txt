export behavior WrappedDegrees for u16 {
  var degrees: u16

  init(initialValue: fn(): u16) { degrees = initialValue() % 360_u16 }

  get {
    defer { degrees %= 360_u16 }
    return degrees
  }
  mut set(newValue) { degrees = newValue % 360_u16 }

  export mut fn reset() {
    degrees = 0
  }
}

export behavior Versioned<Value> for Value {
  var epoch: u64

  init() {
    epoch = 0
  }

  export let mutationEpoch: u64 {
    get => epoch
  }

  willSet(current: ref Value, proposed: ref Value) { }

  mut didSet(current: ref Value) {
    epoch += 1
  }

}

export behavior VersionedDegrees for u16 =
  (degrees: WrappedDegrees, version: Versioned)

export struct Attitude {
  var VersionedDegrees yaw: u16 = 0
}

fn use(
  ovens: ref OvenService,
  store: ref Store,
  task: take Task<Outcome>,
  items: take Array<Item>,
  recipe: Recipe,
  bytes: Bytes,
): Result {
  var attitude = Attitude()
  attitude.yaw = 350_u16
  attitude.yaw += 25_u16
  expect attitude.yaw == 15_u16
  let normalized = copy attitude.yaw |> clamp(0, 360) |> inspect()
  let asyncResult = normalized |> async decode()
  let decoded = bytes |> try json.decode<Document>()
  let transaction = 1
  let raw = #"raw#facet"#
  let ready = try await pipeline ovens.acquire(recipe).preheat()
  let committed = try await pipeline {
    let acquired = ovens.acquire(recipe)
    let prepared = acquired.preheat()
    commit prepared
  }
  let receipt = try await pipeline<transaction: {
    isolation: .readCommitted,
    access: .readWrite,
  }> tx = store {
    let value = tx.read()
    commit value
  }
  let defaultReceipt = try await pipeline<transaction: .default> tx = store {
    commit tx.read()
  }
  let outputs = try await pipeline<
    tasks: .concurrent,
    limit: 16,
    ordering: .input,
    errors: .collect,
  > each item in take items {
    commit item |> inspect()
  }
  let epoch = attitude.yaw#version.mutationEpoch
  expect epoch == 2
  attitude.yaw#version.resetMutationEpoch()
  expect attitude.yaw#version.mutationEpoch == 0
  attitude.yaw#degrees.reset()
  expect attitude.yaw == 0_u16
  expect attitude.yaw#version.mutationEpoch == 1
  let outcome = await (take task)#outcome()
  close(ready, committed, receipt, defaultReceipt, outputs, epoch, outcome, asyncResult, decoded)
}

fn pipePrecedence(x: Value?, fallback: Value, f: fn(Value): Value): Value {
  let left = (x |> f()) ?? fallback
  let right = (x ?? fallback) |> f()
  return right
}

fn invalidPipeRhs(x: Value, fallback: Value): Value {
  // Tree-sitter accepts this CST supergrammar shape; semantic checking rejects
  // the unparenthesized binary suffix because RHS must be one call template.
  let badCoalescing = x |> f() ?? fallback
  let badLogical = x |> f() || fallback
  return badCoalescing
}

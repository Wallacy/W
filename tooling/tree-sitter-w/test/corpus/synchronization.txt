==================
Atomic contracts and scoped locks
==================

enum GateState {
  idle
  busy
}

object Gate {
  var atomic state: GateState = .idle
  var atomic count: u64 = 0

  fn record() {
    count.saturatingAdd<.relaxed>(1)
  }

  fn enter(): AtomicExchange<GateState> {
    return state.compareExchange<
      success: .acquireRelease,
      failure: .acquire,
    >(
      expected: .idle,
      desired: .busy,
    )
  }
}

fn nativeCounterIsLockFree(): Bool {
  return Atomic<u64>.isLockFree
}

fn requiredNativeCounter(): Atomic<u64, lockFree: true> {
  return Atomic<u64, lockFree: true>(0)
}

fn snapshot(lock: ref Mutex<GateState>): GateState {
  return lock.withLock((value: ref GateState) => value)
}

async fn close(lock: ref AsyncMutex<GateState>): GateState {
  return await lock.withLock((value: inout GateState) => {
    value = .idle
    return value
  })
}

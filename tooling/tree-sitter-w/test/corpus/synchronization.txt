Atomic contracts and language lock
==================

enum GateState {
  idle
  busy
}

object Gate {
  var atomic state: GateState = .idle
  var atomic count: u64 = 0

  fn record() {
    count.saturatingAdd<.relaxed>(1)
  }

  fn enter(): AtomicExchange<GateState> {
    return state.compareExchange<
      success: .acquireRelease,
      failure: .acquire,
    >(
      expected: .idle,
      desired: .busy,
    )
  }
}

fn nativeCounterIsLockFree(): Bool {
  return Atomic<u64>.isLockFree
}

fn requiredNativeCounter(): Atomic<u64, lockFree: true> {
  return Atomic<u64, lockFree: true>(0)
}

fn snapshot(state: shared GateState): GateState {
  return lock state as value { value }
}

async fn close(state: shared GateState): GateState {
  return await lock state as value {
    value = .idle
    value
  }
}

fn trySnapshot(state: shared GateState): LockAttempt<GateState> {
  return try lock state as value { value }
}

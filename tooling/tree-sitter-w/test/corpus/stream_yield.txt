fn orderFeed(source: take some Stream<Order, Never>): some Stream<Order, Never> {
  return stream <[take source]> {
    var input = take source
    while let order = await input.next() {
      yield take order
    }
  }
}

fn copyFeed(source: take some Stream<String, Never>): some Stream<String, Never> {
  return stream <[take source]> {
    var input = take source
    while let line = await input.next() {
      yield copy line
    }
  }
}

async fn makeCake(request: CakeRequest): Cake throws KitchenError {
  let plan = try validateCake(request)
  let ingredients = async pantry.fetch(plan.ingredients)
  let ready = try await ingredients
  let cook = spawn<.compute> bake(take ready)
  return try await cook
}

fn containsRatio(value: f64): Bool {
  let closed = value in 0.0...1.0
  let upperOpen = value in 0.0..<1.0
  let lowerOpen = value in 0.0>..1.0
  let open = value in 0.0>..<1.0
  return closed && upperOpen && lowerOpen && open
}

fn shouldAccumulate(raw: f64, error: f64): Bool {
  switch raw {
    case 0.0...1.0:
      return true
    case ..<0.0 if error > 0.0:

    case 1.0>.. if error < 0.0:
      return true
    case _:
      return false
  }

fn fold(rows: ref Array<Array<u8>>): u32 {
  var bits: u32 = 0
  assemble: {
    scanRows: for ref row in rows {
      for value in row {
        bits <<= 5
        bits |= value
        if value == 0 { continue scanRows }
        if value > 31 { break assemble }
      }
    }
  }
  return bits
}

fn digitCount(value: u64): u8 {
  var remaining = value
  var digits: u8 = 0
  repeat {
    digits += 1
    remaining /= 10
  } while remaining > 0
  return digits
}

async fn makeCake(request: CakeRequest): Cake throws KitchenError {
  let plan = try validateCake(request)
  async let ingredients = pantry.fetch(plan.ingredients)
  let ready = try await ingredients
  spawn let cook = bake(take ready)
  return try await cook
}

async fn place(item: MenuItem): Receipt throws MenuError {
  switch item {
    case .cake(let request):
      return try await makeCake(request)
    case .soup(let request):
      return try await makeSoup(request)
  }
}

async fn convert(): Receipt throws MenuError {
  do {
    return try await makeOrder()
  } catch let error {
    throw .order(error)
  }
}

fn containsRatio(value: f64): Bool {
  let closed = value in 0.0...1.0
  let upperOpen = value in 0.0..<1.0
  let lowerOpen = value in 0.0>..1.0
  let open = value in 0.0>..<1.0
  return closed && upperOpen && lowerOpen && open
}

fn shouldAccumulate(raw: f64, error: f64): Bool {
  switch raw {
    case 0.0...1.0:
      return true
    case ..<0.0 if error > 0.0:
      return true
    case 1.0>.. if error < 0.0:
      return true
    case _:
      return false
  }
}

==================
String owns construction and storage reuse
==================

fn build(lines: ref Array<String>, memory: ref Allocator): String throws AllocationError {
  var output = String(using: memory)
  try output.tryReserve(minimumBytes: 4096)

  for ref line in lines {
    output.append(line)
  }

  return output
}

fn drain(buffer: inout String): String {
  return buffer.takeAll()
}

fn bytes(value: take String): Bytes {
  return (take value).intoBytes()
}

fn join(left: take String, right: view String): String {
  return (take left) + right
}

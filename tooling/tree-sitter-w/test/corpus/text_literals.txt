fn literals(): Bytes {
  let normal = "order ${id}\u{2026}"
  let raw = #"C:\orders\${id}"#
  let rawInterpolated = #"{"value":#${id},"unit":"s"}"#
  let multiline = """broth
cake"""
  let rawMultiline = #"""${literal}
C:\last-light"""#
  let scalar = 'λ'
  let byte = b'\xFF'
  return b"WPKG\x00\x01"
}

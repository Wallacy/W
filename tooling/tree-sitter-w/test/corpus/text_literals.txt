fn literals(): Bytes {
  let normal = "order ${id}\u{2026}"
  let single = 'order ${id}\u{2026}'
  let raw = #"C:\orders\${id}"#
  let rawSingle = #'C:\orders\${id}'#
  let multiline = """broth
cake"""
  let singleMultiline = '''broth
cake'''
  let rawMultiline = #"""${literal}
C:\last-light"""#
  let rawSingleMultiline = #'''${literal}
C:\last-light'''#
  let scalar: UnicodeScalar = 'λ'
  let byte = b'\xFF'
  return b"WPKG\x00\x01"
}

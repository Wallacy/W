  completionCount: u64 { get }
  var limit: usize { get set get mut ref }

export struct Counter: CompletionMetric {
  var storedValue: u64

  export init(initial: u64) throws RangeError {
    self.storedValue = initial
  }

  export completionCount: u64 {
    get => storedValue
  }

  var value: u64 {
    get => storedValue
    set(next) => storedValue = next

    get mut ref {
      return mut ref storedValue
    }
  }
}

object Stream {
  var buffer: String

  take fn finish(): String {
    return buffer.takeAll()
  }

  take async fn release() throws CloseError {
    await close(buffer)
  }
}

fn drain(source: take Stream): String {
  return (take source).finish()
}

async fn closeStream(cursor: take Stream) throws CloseError {
  try await (take cursor).release()
}

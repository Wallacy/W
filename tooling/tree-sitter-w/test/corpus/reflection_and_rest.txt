==================
Reflection and homogeneous rest arguments
==================

struct Ticket: Hashable & Reflectable {
  id: u64
}

enum Signal: Reflectable {
  ready(Ticket)
  delayed(u32)
  stopped
}

alias ActiveSignal = Signal<[.ready, .delayed]>

fn load(kitchens: u16, courses: Course...): u32 {
  return courses.count * kitchens
}

fn announce(_ messages: ref String...): usize {
  return messages.count
}

fn archive(_ records: take Record...): usize {
  return records.count
}

fn inspect(value: ref any Reflectable, other: TypeId) {
  let planned = [.broth, .cake]
  let count = load(2, courses: .salad, each planned)
  let tail = planned.slice(4...)
  let staticId = type of Ticket
  let T = Ticket(id: 1)
  let staticHomonym = type of T
  let dynamicHomonym = type of (T)
  let dynamicId = type of value
  let ref metadata = info of ActiveSignal
  let ref dynamicMetadata = info of value
  let recovered = value as? Ticket
  let queryPrecedence = type of value == other
  let groupedQuery = type of (value == other)
  let loader: fn(u16, Course...): u32 = load
}

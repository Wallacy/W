import reflect from std

struct Ticket: Hashable & reflect.Reflectable {
  id: u64
}

enum Signal: reflect.Reflectable {
  ready(Ticket)
  delayed(u32)
  stopped
}

alias ActiveSignal = Signal<[.ready, .delayed]>

fn load(kitchens: u16, courses: Course...): u32 {
  return courses.count * kitchens
}

fn announce(_ messages: ref String...): usize {
  return messages.count
}

fn archive(_ records: take Record...): usize {
  return records.count
}

fn inspect() {
  let planned = [.broth, .cake]
  let count = load(2, courses: .salad, each planned)
  let tail = planned.slice(4...)
  let id = reflect.TypeId.of<Ticket>()
  let ref info = reflect.info<ActiveSignal>()
  let loader: fn(u16, Course...): u32 = load
}

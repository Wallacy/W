fn bind(source: ref Source) {
  let (head, tail) = source.split()
  if let (first, rest) = source.nextPair() {
    inspect(first, rest)
  }
  guard let Ticket(id, state, ...) = source.nextTicket() else return
  while let (frame, remaining) = source.nextFrame() {
    consume(frame, remaining)
  }
  for (index, value) in source.enumerated() {
    record(index, value)
  }
}

fn requested(reading: OvenReading, stage: ServiceStage): Temperature {
  switch stage {
    case ServiceStage.preparing: prepare()
    case .serving: serve()
    case _: wait()
  }

  return switch reading {
    case .stable(let temperature): temperature
    case .warming(target: let target, ...): target
    case .failed(let error): throw error
  }
}

fn band(value: Int<(0...9)>): Band {
  return switch value {
    case 0...2: .low
    case 3...7: .middle
    case 8...9: .high
  }
}

fn isSafe(temperature: Temperature): Bool {
  return switch temperature {
    case 30<degC>...300<degC>: true
    case _: false
  }
}

fn inspect(ticket: ref Ticket): State {
  return switch ticket {
    case Ticket(state: .open, attempts: 0...3, ...): .active
    case Ticket(state: .closed, ...): .done
    case _: .invalid
  }
}

protocol Stream<Item, Failure: Error> {
  mut async fn next(): Item? throws Failure
}

async fn consume(
  source: take some Stream<Order, StreamError>,
): () throws StreamError {
  var orders = take source

  for try await order in orders {
    inspect(order)
  }

  while let order = try await orders.next() {
    inspect(order)
  }
}

Directional channel contracts

async fn submit(
  output: ref Channel<Order><.send>,
  order: take Order,
): () throws ChannelSendError<Order><[.closed]> {
  try await output.send(take order)
}

async fn drain(
  input: take Channel<Order><.receive>,
) {
  var orders = take input

  for await order in orders {
    serve(take order)
  }
}

==================
Service stream positions and relay pipeline
==================

protocol TelemetryApi {
  async fn follow(
    after sequence: u64,
  ): some Stream<Telemetry, TelemetryError>
}

protocol ArchiveApi {
  async fn ingest(
    source: take some Stream<Telemetry, TelemetryError>,
  ): Receipt throws ArchiveError
}

async fn relay(
  telemetry: ServiceRef<TelemetryApi>,
  archive: ServiceRef<ArchiveApi>,
  sequence: u64,
): Receipt throws RelayError {
  return try await pipeline {
    let feed = telemetry.follow(after: sequence)
    let receipt = archive.ingest(source: take feed)
    return receipt
  }
}

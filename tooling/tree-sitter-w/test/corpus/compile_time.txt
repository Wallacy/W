const fn buildOpcodes(): Map<String, u8> {
  var result = Map<String, u8>()
  result["heat"] = 0x02_u8
  result["serve"] = 0xff_u8
  return result
}

const opcodes = buildOpcodes()

struct Cell {
  var value: u8

  const init(value: u8) {
    self.value = value
  }

  const mut fn replace(value: u8) {
    self.value = value
  }

  const take fn finish(): u8 {
    return value
  }

  static const fn zero(): Cell {
    return Cell(value: 0)
  }
}

protocol ConstDecodable {
  static const fn decode(source: ref Bytes): Self throws DecodeError
}

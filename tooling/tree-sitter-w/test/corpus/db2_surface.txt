fn classify<const batch: usize>(
  input: ref Tensor<f32, shape: [batch, 8]>,
  weights: ref Tensor<f32, shape: [8, 4]>,
): Tensor<f32, shape: [batch, 4]> {
  let logits = input @ weights
  return switch batch {
    case 0: logits
    case _: logits
  }
}

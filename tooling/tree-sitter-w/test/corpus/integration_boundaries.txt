fn inspect(rows: Rows, order: Order) {
  scanRows: for ref row in rows {
    for value in row {
      if value == 0 { continue scanRows }
      if value > 31 { break scanRows }
      consume(value)
    }
  }

  validateOrder: {
    if order.isEmpty { break validateOrder }
    reserve(order)
  }
}
